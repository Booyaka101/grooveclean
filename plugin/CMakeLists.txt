cmake_minimum_required(VERSION 3.25)
project(GrooveCleanPlugin VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
find_package(Threads REQUIRED)
add_library(groove_stream STATIC src/StreamEngine.cpp)
target_include_directories(groove_stream PUBLIC src)
target_compile_features(groove_stream PUBLIC cxx_std_17)
target_link_libraries(groove_stream PUBLIC Threads::Threads)

if(BUILD_TESTING)
    add_executable(groove_stream_test tests/stream_test.cpp)
    target_link_libraries(groove_stream_test PRIVATE groove_stream)
    add_test(NAME stream COMMAND groove_stream_test)
    set_tests_properties(stream PROPERTIES TIMEOUT 60)
endif()

option(GROOVECLEAN_BUILD_VST3 "Build the Windows VST3 plugin" ON)
if(NOT GROOVECLEAN_BUILD_VST3)
    return()
endif()
if(NOT WIN32)
    message(FATAL_ERROR "The worker transport currently supports Windows only")
endif()
set(VST3_SDK_ROOT "" CACHE PATH "VST3 SDK 3.8.0 checkout with submodules")
if(NOT EXISTS "${VST3_SDK_ROOT}/public.sdk/source/vst/vstaudioeffect.h")
    message(FATAL_ERROR "Set VST3_SDK_ROOT to a VST3 SDK 3.8.0 checkout")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
smtg_add_vst3plugin(GrooveClean src/Plugin.cpp src/WorkerProcess.cpp src/Editor.cpp)
target_compile_features(GrooveClean PRIVATE cxx_std_17)
target_compile_definitions(GrooveClean PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(GrooveClean PRIVATE "${VST3_SDK_ROOT}")
target_link_libraries(GrooveClean PRIVATE sdk groove_stream comctl32)
if(BUILD_TESTING)
    add_executable(groove_worker_test tests/worker_test.cpp src/WorkerProcess.cpp)
    target_compile_definitions(groove_worker_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(groove_worker_test PRIVATE groove_stream)
    add_test(NAME worker COMMAND groove_worker_test)
    set_tests_properties(worker PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
    add_executable(groove_plugin_test tests/plugin_test.cpp
        "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp"
        "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp")
    target_compile_features(groove_plugin_test PRIVATE cxx_std_17)
    target_compile_definitions(groove_plugin_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(groove_plugin_test PRIVATE "${VST3_SDK_ROOT}")
    target_link_libraries(groove_plugin_test PRIVATE sdk_hosting sdk)
    add_dependencies(groove_plugin_test GrooveClean)
    add_test(NAME plugin COMMAND groove_plugin_test $<TARGET_FILE:GrooveClean>)
    set_tests_properties(plugin PROPERTIES TIMEOUT 60)
endif()
